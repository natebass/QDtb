	if (( $# == 0 )) then 
		ls

		cd "$@" && ls
	fi

	git add -A;
	git commit -m "### QUICK-COMMIT ###";

}

  git checkout master && 
	git fetch origin &&
	git pull origin master &&
	git checkout $1 &&
	git rebase master
}
