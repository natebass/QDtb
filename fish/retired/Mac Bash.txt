# Functions {{{
gitq(){ git add -A;git commit -m '###QUICK COMMIT###';git push;}
chdir(){
	mkdir $1; cd $_;
}
d(){
	if [ $# -eq 0  ]; then
		ls -F
	else
		builtin cd "$@" && ls -Ft
	fi
}
# }}}
# Prompt {{{
blue=`tput setaf 4`

reset=`tput sgr0`
PS1='${blue}$PWD${reset}\n⟩ '
